\section{Definición del problema}
% =====================================================================

\begin{frame}{Un problema en dos niveles articulados}
  \begin{columns}[T]
    \column{.48\textwidth}
      \begin{block}{Nivel 1 -- Asignación}
        Minimizar una función objetivo \textbf{estocástica} con variables continuas
        (proporciones, umbrales de priorización) y enteras (cupos por etapa,
        bloques de agenda), cuya varianza \textbf{depende del punto evaluado}.
      \end{block}
    \column{.48\textwidth}
      \setbeamercolor{block title}{bg=ucorange,fg=white}
      \setbeamercolor{block body}{bg=ucsand}
      \begin{block}{Nivel 2 -- Coordinación}
        Un mecanismo entre dos instituciones que reparta el recurso compartido de
        modo que \textbf{ninguna tenga incentivos para operar sola}, incorporando el
        margen de error con que se conoce el desempeño de cada una.
      \end{block}
  \end{columns}
  \vspace{1.5em}
  \begin{center}
    \textit{No existe un marco analítico que apoye de manera integrada las decisiones
    de capacidad bajo incertidumbre.}
  \end{center}
\end{frame}

\begin{frame}{Seis rasgos que el método debe respetar}
  \begin{itemize}
    \item<1-> \textbf{Demanda estocástica:} llegadas, inasistencias y cancelaciones.
    \item<2-> \textbf{Espacio de decisión entero-mixto:} cupos enteros y proporciones continuas.
    \item<3-> \textbf{Ruido heterocedástico estructural:} $\sigma^2(x)$ depende del régimen operativo.
    \item<4-> \textbf{Evaluaciones costosas:} cada configuración es una corrida DES.
    \item<5-> \textbf{Sin gradiente:} el simulador es una caja negra.
    \item<6-> \textcolor{ucorange}{\textbf{Varias instituciones:}} objetivos propios sobre un recurso compartido.
  \end{itemize}
\end{frame}

\section{Revisión bibliográfica}
% =====================================================================

\begin{frame}{Pilares fundamentales}
  \begin{itemize}
    \item<1-> \textbf{Simulación de eventos discretos (DES):} representa flujos de
      pacientes, colas y reglas operativas, pero se usa sobre todo para evaluar
      escenarios predefinidos (Law, 2015; Monks y Harper, 2025).
    \item<2-> \textbf{Optimización vía simulación (SO) y DFO:} ASTRO-DF, COMPASS,
      R-SPLINE, SMAC3 y regiones de confianza con GP; garantías bajo varianza finita
      y homogénea o en espacios continuos (Shashaani et al., 2018; Hong y Nelson, 2006;
      Lindauer et al., 2022).
    \item<3-> \textbf{Metaheurísticas} (SA, GA, OptQuest): frecuentes en salud, sin
      garantías cuando la varianza depende del punto (Wang y Demeulemeester, 2023).
    \item<4-> \textbf{Teoría de juegos cooperativa:} núcleo, Shapley y Nash para
      pooling de capacidad (Karsten et al., 2015; Gao et al., 2024); con valores de
      coalición inciertos tomados como dato (Li y Conitzer, 2015; Pantazis et al., 2022, 2024).
  \end{itemize}
\end{frame}

\section{Resultados}

\begin{frame}{Resultados preliminares (IFORS 2026): los algoritmos superan toda política manual}
  \begin{columns}[T]
    \column{.48\textwidth}\bignum{263 $\rightarrow$ 177 días}{tiempo promedio en sistema con SMAC-SK ($-33$\,\%)}
    \column{.48\textwidth}\bignum{1.148 $\rightarrow$ 2.179}{pacientes atendidos ($+90$\,\%), sin trade-off}
  \end{columns}
  \vspace{1em}
  \begin{itemize}
    \item<2-> \textbf{Los 7 algoritmos} reducen significativamente el tiempo en sistema
      (Welch, $p \ll 0{,}001$); benchmark de $7\times10$ semillas con números aleatorios comunes.
    \item<3-> \textcolor{ucorange}{\textbf{Pero las garantías no se transfieren:}} suponen
      ruido homocedástico y suavidad, ambas violadas por el evaluador DES
      $\Rightarrow$ motiva OE2 y OE3.
  \end{itemize}
  \vfill
  {\scriptsize Araneda y Carrasco (2026). IFORS, Viena.}
\end{frame}

\begin{frame}{Resultados esperados}
  \begin{itemize}
    \item Un \textbf{modelo DES calibrado y validado}, con la estructura heterocedástica
      de su evaluador caracterizada.
    \item Un \textbf{algoritmo DFO entero-mixto} para evaluadores heterocedásticos con
      asignación adaptativa de réplicas.
    \item \textbf{Evidencia comparativa} frente al estado del arte: calidad, eficiencia
      y estabilidad bajo distintos niveles de heterocedasticidad.
    \item Un \textbf{mecanismo de gobernanza robusto} y las condiciones bajo las cuales
      la coordinación es estable y mutuamente conveniente.
    \item Un \textbf{marco prescriptivo integrado} en un prototipo computacional,
      transferible a otros servicios de especialidad de la red.
  \end{itemize}
\end{frame}
